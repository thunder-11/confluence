cmake_minimum_required(VERSION 3.20)
project(confluence VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(confluence_core INTERFACE)
target_include_directories(confluence_core INTERFACE engine/include engine/vendor)
add_executable(confluence_engine engine/src/main.cpp)
target_link_libraries(confluence_engine PRIVATE confluence_core)
add_executable(confluence_tests engine/tests/tests.cpp)
target_link_libraries(confluence_tests PRIVATE confluence_core)
enable_testing()
add_test(NAME algorithms COMMAND confluence_tests)
